"use client";

import React, { useState, useMemo } from 'react';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';

/**

* Perspectiva Matemática SOTA:
* O cálculo matemático puro de pathfinding geométrico para isolar a fração (f) necessária.
* Formula: (1 + 2f)^n = target_pot / current_pot
 */
function calculateGeometricSizing(currentPot: number, targetPot: number, remainingStreets: number): number {
    if (currentPot <= 0 || targetPot <= currentPot || remainingStreets <= 0) return 0.0;
    const growthFactor = targetPot / currentPot;
    const onePlusTwoF = Math.pow(growthFactor, 1.0 / remainingStreets);
    return (onePlusTwoF - 1.0) / 2.0;
}

export default function GtoCfrLaboratory() {
    const [currentPot, setCurrentPot] = useState<number>(100);
    const [targetPot, setTargetPot] = useState<number>(1000);
    const [streets, setStreets] = useState<number>(3); // Ex: Flop, Turn, River

    // Lei de Shannon (Zero-Rework): Computação e memoização da árvore geométrica
    const { fraction, progressionData } = useMemo(() => {
        const f = calculateGeometricSizing(currentPot, targetPot, streets);
        const data = [];
        let tempPot = currentPot;

        for (let i = 1; i <= streets; i++) {
            const betSize = tempPot * f;
            const potAntes = tempPot;
            tempPot += betSize * 2; // Simetria de crescimento sistêmico (Aposta + Call)

            data.push({
                street: i === 1 ? 'Flop' : i === 2 ? 'Turn' : i === 3 ? 'River' : `St ${i}`,
                potAntes: Number(potAntes.toFixed(2)),
                aposta: Number(betSize.toFixed(2)),
                potDepois: Number(tempPot.toFixed(2)),
                fractionLabel: `${(f * 100).toFixed(1)}%`
            });
        }

        return { fraction: f, progressionData: data };
    }, [currentPot, targetPot, streets]);

    return (
        <div className="flex flex-col w-full h-full p-8 bg-zinc-950 text-zinc-100 space-y-8 border border-zinc-800 rounded-xl font-mono">
            <header className="space-y-2">
                <h1 className="text-3xl font-black text-zinc-50 tracking-tight">CFR LAB: Geometric Sizing Pathfinding</h1>
                <p className="text-sm text-zinc-400">Distribuição estrita da taxa de stack-to-pot (SPR) para maximização de EV.</p>
            </header>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
                <div className="flex flex-col space-y-2 p-5 bg-zinc-900 border border-zinc-800 rounded-lg">
                    <label className="text-xs font-bold text-zinc-500 uppercase tracking-widest">Current Pot (Pote Inicial)</label>
                    <input
                        type="number"
                        value={currentPot}
                        onChange={(e) => setCurrentPot(Number(e.target.value))}
                        className="bg-zinc-950 border border-zinc-700 rounded p-2 text-zinc-100 focus:border-cyan-500 focus:ring-1 focus:ring-cyan-500 outline-none transition-all"
                    />
                </div>
                <div className="flex flex-col space-y-2 p-5 bg-zinc-900 border border-zinc-800 rounded-lg">
                    <label className="text-xs font-bold text-zinc-500 uppercase tracking-widest">Target Pot (Stack Efetivo Alvo)</label>
                    <input
                        type="number"
                        value={targetPot}
                        onChange={(e) => setTargetPot(Number(e.target.value))}
                        className="bg-zinc-950 border border-zinc-700 rounded p-2 text-zinc-100 focus:border-cyan-500 focus:ring-1 focus:ring-cyan-500 outline-none transition-all"
                    />
                </div>
                <div className="flex flex-col space-y-2 p-5 bg-zinc-900 border border-zinc-800 rounded-lg">
                    <label className="text-xs font-bold text-zinc-500 uppercase tracking-widest">Streets Restantes</label>
                    <input
                        type="number"
                        value={streets}
                        min={1} max={4}
                        onChange={(e) => setStreets(Number(e.target.value))}
                        className="bg-zinc-950 border border-zinc-700 rounded p-2 text-zinc-100 focus:border-cyan-500 focus:ring-1 focus:ring-cyan-500 outline-none transition-all"
                    />
                </div>
            </div>

            <div className="flex flex-col p-6 bg-zinc-900 border border-zinc-800 rounded-lg shadow-xl">
                <div className="mb-8 flex justify-between items-center border-b border-zinc-800 pb-4">
                    <h2 className="text-xl font-bold text-zinc-200">Vetores de Colisão Pós-Flop</h2>
                    <div className="px-5 py-2 bg-cyan-950/40 border border-cyan-800 rounded-md">
                        <span className="text-sm font-bold text-cyan-400">
                            Fração Estática (f): {(fraction * 100).toFixed(2)}% do Pote
                        </span>
                    </div>
                </div>

                <div className="h-96 w-full">
                    <ResponsiveContainer width="100%" height="100%">
                        <BarChart data={progressionData} margin={{ top: 20, right: 30, left: 0, bottom: 0 }}>
                            <CartesianGrid strokeDasharray="3 3" stroke="#27272a" vertical={false} />
                            <XAxis dataKey="street" stroke="#71717a" tick={{ fill: '#a1a1aa' }} axisLine={false} tickLine={false} />
                            <YAxis stroke="#71717a" tick={{ fill: '#a1a1aa' }} axisLine={false} tickLine={false} />
                            <Tooltip
                                cursor={{ fill: '#18181b' }}
                                contentStyle={{ backgroundColor: '#09090b', borderColor: '#27272a', color: '#f4f4f5', borderRadius: '8px' }}
                                itemStyle={{ color: '#22d3ee', fontWeight: 'bold' }}
                            />
                            {/* Antevisão Semântica do Stack Embutida Visualmente */}
                            <Bar dataKey="potAntes" name="Pote (Base)" stackId="a" fill="#27272a" radius={[0, 0, 4, 4]} />
                            <Bar dataKey="aposta" name="Investimento Otimizado (Hero)" stackId="a" fill="#06b6d4" />
                            <Bar dataKey="aposta" name="Aprovação (Villain Call)" stackId="a" fill="#0891b2" radius={[4, 4, 0, 0]} />
                        </BarChart>
                    </ResponsiveContainer>
                </div>
            </div>
        </div>
    );
}
